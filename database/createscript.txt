-- Toolly database (Year 11 version)
-- One table per main feature: accounts, listings, bookings, wishlists,
-- reviews, claims, and messages.

CREATE TABLE users (
    userid INTEGER PRIMARY KEY AUTOINCREMENT,
    firstname TEXT NOT NULL,
    lastname TEXT NOT NULL,
    email TEXT NOT NULL UNIQUE,
    password TEXT NOT NULL,
    permission TEXT NOT NULL CHECK (permission IN ('admin', 'User (Renter)', 'User (Tool Provider)')),
    lastaccess DATE NOT NULL DEFAULT (datetime('now','localtime')),
    status TEXT,
    profilephoto TEXT
);

CREATE TABLE tools (
    toolid INTEGER PRIMARY KEY AUTOINCREMENT,
    providerid INTEGER NOT NULL,
    title TEXT NOT NULL,
    description TEXT NOT NULL,
    daily_rate REAL NOT NULL CHECK (daily_rate > 0),
    original_value REAL NOT NULL CHECK (original_value > 0),
    city TEXT NOT NULL,
    suburb TEXT NOT NULL,
    location TEXT NOT NULL,
    tool_type TEXT NOT NULL,
    brand TEXT NOT NULL,
    tool_condition TEXT NOT NULL,
    toolphoto TEXT NOT NULL,
    available_from DATE NOT NULL,
    available_until DATE NOT NULL,
    is_available INTEGER NOT NULL DEFAULT 1 CHECK (is_available IN (0, 1)),
    created_at DATE NOT NULL DEFAULT (datetime('now','localtime')),
    FOREIGN KEY (providerid) REFERENCES users(userid)
);

CREATE TABLE tool_rentals (
    rentalid INTEGER PRIMARY KEY AUTOINCREMENT,
    toolid INTEGER NOT NULL,
    renterid INTEGER NOT NULL,
    providerid INTEGER NOT NULL,
    start_date DATE NOT NULL,
    end_date DATE NOT NULL,
    rental_days INTEGER NOT NULL,
    rental_cost REAL NOT NULL,
    security_deposit REAL NOT NULL,
    insurance_selected INTEGER NOT NULL DEFAULT 0,
    insurance_cost REAL NOT NULL DEFAULT 0,
    total REAL NOT NULL,
    status TEXT NOT NULL DEFAULT 'active' CHECK (status IN ('active', 'completed')),
    created_at DATE NOT NULL DEFAULT (datetime('now','localtime')),
    completed_at DATE,
    FOREIGN KEY (toolid) REFERENCES tools(toolid),
    FOREIGN KEY (renterid) REFERENCES users(userid),
    FOREIGN KEY (providerid) REFERENCES users(userid)
);

CREATE TABLE tool_wishlists (
    wishlistid INTEGER PRIMARY KEY AUTOINCREMENT,
    renterid INTEGER NOT NULL,
    toolid INTEGER NOT NULL,
    UNIQUE (renterid, toolid),
    FOREIGN KEY (renterid) REFERENCES users(userid),
    FOREIGN KEY (toolid) REFERENCES tools(toolid)
);

CREATE TABLE reviews (
    reviewid INTEGER PRIMARY KEY AUTOINCREMENT,
    rentalid INTEGER NOT NULL,
    authorid INTEGER NOT NULL,
    rating INTEGER NOT NULL CHECK (rating BETWEEN 1 AND 5),
    comment TEXT,
    created_at DATE NOT NULL DEFAULT (datetime('now','localtime')),
    UNIQUE (rentalid, authorid),
    FOREIGN KEY (rentalid) REFERENCES tool_rentals(rentalid),
    FOREIGN KEY (authorid) REFERENCES users(userid)
);

CREATE TABLE claims (
    claimid INTEGER PRIMARY KEY AUTOINCREMENT,
    rentalid INTEGER NOT NULL,
    providerid INTEGER NOT NULL,
    description TEXT NOT NULL,
    evidence_photo TEXT NOT NULL,
    claim_charge REAL NOT NULL DEFAULT 0,
    status TEXT NOT NULL DEFAULT 'open' CHECK (status IN ('open', 'reviewing', 'accepted', 'denied')),
    created_at DATE NOT NULL DEFAULT (datetime('now','localtime')),
    reviewed_at DATE,
    FOREIGN KEY (rentalid) REFERENCES tool_rentals(rentalid),
    FOREIGN KEY (providerid) REFERENCES users(userid)
);

CREATE TABLE messages (
    messageid INTEGER PRIMARY KEY AUTOINCREMENT,
    toolid INTEGER NOT NULL,
    senderid INTEGER NOT NULL,
    recipientid INTEGER NOT NULL,
    messagetext TEXT NOT NULL,
    created_at DATE NOT NULL DEFAULT (datetime('now','localtime')),
    FOREIGN KEY (toolid) REFERENCES tools(toolid),
    FOREIGN KEY (senderid) REFERENCES users(userid),
    FOREIGN KEY (recipientid) REFERENCES users(userid)
);

CREATE INDEX idx_tools_search ON tools(city, suburb, is_available);
CREATE INDEX idx_rentals_renter ON tool_rentals(renterid, status);
CREATE INDEX idx_rentals_provider ON tool_rentals(providerid, status);
CREATE INDEX idx_claims_provider ON claims(providerid, status);
CREATE INDEX idx_messages_user ON messages(recipientid, senderid);
